cmake_minimum_required(VERSION 3.28)
project(MapForGoblinsAdapter LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
FetchContent_Declare(minhook GIT_REPOSITORY https://github.com/TsudaKageyu/minhook.git
    GIT_TAG f5485b8454544c2f034c78f8f127c1d03dea3636)
FetchContent_MakeAvailable(minhook)
add_library(MapForGoblins SHARED adapter.cpp)
target_compile_definitions(MapForGoblins PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(MapForGoblins PRIVATE /W4 /WX /EHsc)
target_link_libraries(MapForGoblins PRIVATE minhook bcrypt)
add_executable(adapter_model_test ../tests/adapter_model_test.cpp)
target_compile_options(adapter_model_test PRIVATE /W4 /WX /UNDEBUG)
enable_testing()
add_test(NAME adapter_model COMMAND adapter_model_test)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
    add_test(NAME adapter_rejects_unknown_release COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/../tests/adapter_rejection_test.py" "$<TARGET_FILE:MapForGoblins>")
endif()
option(MFG_BUILD_SMOKE_CLIENT "Build separate offline API test DLL (never package for players)" OFF)
if(MFG_BUILD_SMOKE_CLIENT)
    add_library(adapter_smoke_client SHARED ../tests/adapter_smoke_client.cpp)
    target_compile_options(adapter_smoke_client PRIVATE /W4 /WX /EHsc)
endif()
